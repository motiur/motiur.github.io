Answer shape: the answer in one sentence, then the reason, then an example or a caveat. Stop at 90 seconds.

import random
QUESTIONS = {
 "generalization": ["Why can a bigger model generalize better, not worse?", "What is the difference between validation and test data?",
                    "What is the bias-variance trade-off?", "How do you detect overfitting?"],
 "optimization": ["Why does Adam often need less tuning than SGD?", "What causes exploding gradients, and two fixes?",
                  "What does momentum do?", "Why warm up the learning rate?"],
 "regularization": ["Why does L1 give sparse weights and L2 not?", "How does dropout behave differently at test time?",
                    "How is early stopping a regularizer?", "What does weight decay do?"],
 "metrics": ["When would you prefer precision-recall to ROC curves?", "Your classes are 99 to 1. What do you report?",
             "What is macro-F1, and when do you use it?", "What does AUC measure?"],
 "probability": ["Why is minimizing cross-entropy maximum likelihood?", "What is the variance of the mean of n independent draws?",
                 "State Bayes' rule and give an example.", "What is a confidence interval?"],
 "practice": ["Training loss falls, validation loss rises. What now?", "Loss is NaN at step 300. What do you check?",
              "Your result is too good to be true. What do you check?", "How do you choose a learning rate?"],
}
random.seed(0)
drawn = [(topic, q) for topic, qs in QUESTIONS.items() for q in qs]
random.shuffle(drawn)
for i, (topic, q) in enumerate(drawn[:20], 1):
    print(f"{i:2d}. [{topic}] {q}")

Worked numbers to know¶

# Bayes: prevalence 1%, sensitivity 99%, specificity 95%
prev, sens, spec = 0.01, 0.99, 0.95
p_pos = prev * sens + (1 - prev) * (1 - spec)
print("P(disease | positive) =", round(prev * sens / p_pos, 3))

# Variance of a mean of n draws is sigma^2 / n: check by simulation
import numpy as np
rng = np.random.default_rng(0)
means = rng.normal(0, 2, (20000, 25)).mean(1)
print("simulated variance of the mean:", round(means.var(), 3), "| sigma^2 / n =", 4 / 25)
